CREATE VIEW View_Autorizacao_Acesso_Cliente     
AS    
SELECT       
	N2.Id_Grupo,      
	N2.Cod_Grupo,      
	N2.Id_Cliente,      
	CLI2.Tp_Veiculo,    
	CLI2.Id_RM,    
	Nm_RM = RM.Nm_Colaborador,    
	CLI2.Id_Investor,    
	Nm_Investor = INV.Nm_Colaborador,    
	CLI2.Id_Gestor,    
	Nm_Gestor = GES.Nm_Colaborador,    
	CLI2.Id_Conta,    
	CLI2.Cod_Conta,    
	CLI2.Id_Ativo_Fundo_Interno,    
	CLI2.Nr_Cnpj_Fundo,    
	CLI2.Cod_ISIN    
FROM      
(      
	SELECT DISTINCT       
		AG.Id_Grupo,       
		AG.Cod_Grupo,       
		Id_Cliente = COALESCE(CLI_RM.Id_Cliente, CLI_INVESTOR.Id_Cliente, CLI_GESTOR.Id_Cliente)          
	FROM Tb_Grupo_Acesso_Cliente GAC      
	JOIN Tb_Autorizacao_Grupo AG 
	ON AG.Id_Grupo = GAC.Id_Grupo      
	LEFT JOIN View_Cliente CLI_RM 
	ON CLI_RM.Id_RM = GAC.Id_Colaborador 
	AND GAC.Tp_Vinculo = 'RM' -- View RM      
	LEFT JOIN View_Cliente CLI_INVESTOR 
	ON CLI_INVESTOR.Id_Investor = GAC.Id_Colaborador 
	AND GAC.Tp_Vinculo = 'INVESTOR' -- View Investor      
	LEFT JOIN View_Cliente CLI_GESTOR 
	ON CLI_GESTOR.Id_Gestor = GAC.Id_Colaborador 
	AND GAC.Tp_Vinculo = 'GESTOR' -- View Gestor      
  
	UNION ALL      
  
	SELECT
		AG.Id_Grupo,       
		AG.Cod_Grupo,       
		EXC.Id_Cliente      
	FROM Tb_Grupo_Acesso_Cliente_Excecao EXC      
	JOIN Tb_Autorizacao_Grupo AG 
	ON AG.Id_Grupo = EXC.Id_Grupo      
) N2      
JOIN View_Cliente CLI2 
ON CLI2.Id_Cliente = N2.Id_Cliente      
LEFT JOIN Tb_Autorizacao_Grupo_Funcionalidade CAN_VIEW_CONTA (NOLOCK) 
ON CAN_VIEW_CONTA.Cod_Funcionalidade = 'CAN_VIEW_CONTA' 
AND CAN_VIEW_CONTA.Id_Grupo = N2.Id_Grupo      
LEFT JOIN Tb_Autorizacao_Grupo_Funcionalidade CAN_VIEW_FUNDO (NOLOCK) 
ON CAN_VIEW_FUNDO.Cod_Funcionalidade = 'CAN_VIEW_FUNDO' 
AND CAN_VIEW_FUNDO.Id_Grupo = N2.Id_Grupo 
LEFT JOIN Tb_Autorizacao_Grupo_Funcionalidade CAN_VIEW_FUNDO_EXTERNO (NOLOCK) 
ON CAN_VIEW_FUNDO_EXTERNO.Cod_Funcionalidade = 'CAN_VIEW_FUNDO_EXTERNO' 
AND CAN_VIEW_FUNDO_EXTERNO.Id_Grupo = N2.Id_Grupo     
LEFT JOIN Tb_Colaborador RM (NOLOCK) 
ON RM.Id_Colaborador = CLI2.Id_RM    
LEFT JOIN Tb_Colaborador INV (NOLOCK) 
ON INV.Id_Colaborador = CLI2.Id_Investor    
LEFT JOIN Tb_Colaborador GES (NOLOCK) 
ON GES.Id_Colaborador = CLI2.Id_Gestor    
WHERE       
(
	(CLI2.Tp_Veiculo = 'CL' AND CAN_VIEW_CONTA.Cod_Funcionalidade IS NOT NULL) 
	OR 
	(CLI2.Tp_Veiculo = 'FI' AND CAN_VIEW_FUNDO.Cod_Funcionalidade IS NOT NULL)
	OR 
	(CLI2.Tp_Veiculo = 'FE' AND CAN_VIEW_FUNDO_EXTERNO.Cod_Funcionalidade IS NOT NULL)
);


CREATE VIEW View_Autorizacao_Acesso_Grupo   
AS  
SELECT
	gr.Cod_Grupo,  
	gr.Desc_Grupo,  
	co.Nm_Colaborador,  
	gac.Tp_Vinculo,  
	acess.Fl_Acessa_Fundo_Interno,  
	acess.Fl_Acessa_Conta,
	acess.Fl_Acessa_Fundo_Externo	
FROM Tb_Grupo_Acesso_Cliente gac  
JOIN Tb_Autorizacao_Grupo gr  
ON gr.Id_Grupo = gac.Id_Grupo  
JOIN Tb_Colaborador co  
ON co.Id_Colaborador = gac.Id_Colaborador  
JOIN ( 	SELECT gru.Id_Grupo  
		, CASE WHEN (	SELECT COUNT(*)   
						FROM Tb_Autorizacao_Grupo_Funcionalidade ff   
						WHERE ff.Id_Grupo = gru.Id_Grupo   
						AND ff.Cod_Funcionalidade = 'CAN_VIEW_FUNDO'
					) > 0 THEN 'SIM' ELSE 'NAO' END Fl_Acessa_Fundo_Interno  
		, CASE WHEN (	SELECT COUNT(*)   
						FROM Tb_Autorizacao_Grupo_Funcionalidade ff   
						WHERE ff.Id_Grupo = gru.Id_Grupo   
						AND ff.Cod_Funcionalidade = 'CAN_VIEW_CONTA'
					) > 0 THEN 'SIM' ELSE 'NAO' END Fl_Acessa_Conta  
		, CASE WHEN (	SELECT COUNT(*)   
						FROM Tb_Autorizacao_Grupo_Funcionalidade ff   
						WHERE ff.Id_Grupo = gru.Id_Grupo   
						AND ff.Cod_Funcionalidade = 'CAN_VIEW_FUNDO_EXTERNO'
					) > 0 THEN 'SIM' ELSE 'NAO' END Fl_Acessa_Fundo_Externo  
		FROM Tb_Autorizacao_Grupo gru  
	) acess 
ON acess.Id_Grupo = gr.Id_Grupo


CREATE VIEW View_Autorizacao_Grupo_Funcionalidade
AS    
SELECT 
	grp.Cod_Grupo,
	grp.Desc_Grupo,
	grp_func.Cod_Funcionalidade,
	func.Desc_Funcionalidade,
	grp_func.Id_Usu,
	grp_func.Dt_Cpu
FROM Tb_Autorizacao_Grupo_Funcionalidade grp_func
JOIN Tb_Autorizacao_Grupo grp
ON grp.Id_Grupo = grp_func.Id_Grupo
JOIN Tb_Autorizacao_Funcionalidade_Api func_api
ON func_api.Cod_Funcionalidade = grp_func.Cod_Funcionalidade
JOIN Tb_Autorizacao_Funcionalidade func
ON func.Cod_Funcionalidade = grp_func.Cod_Funcionalidade


CREATE VIEW View_Autorizacao_Usuario_Externo  
AS    
SELECT 
	usu.Client_Id,
	usu.Client_Nick,
	grp.Cod_Grupo,
	grp_func.Cod_Funcionalidade,
	func.Desc_Funcionalidade,
	api.Url_Api,
	api.Desc_Api,
	Id_Usu = IIF(grp_func.Dt_Cpu > func_api.Dt_Cpu, grp_func.Id_Usu, func_api.Id_Usu),
	Dt_Cpu = IIF(grp_func.Dt_Cpu > func_api.Dt_Cpu, grp_func.Dt_Cpu, func_api.Dt_Cpu)
FROM Tb_Autorizacao_Grupo_Funcionalidade grp_func
JOIN Tb_Autorizacao_Grupo grp
ON grp.Id_Grupo = grp_func.Id_Grupo
JOIN Tb_Autorizacao_Funcionalidade_Api func_api
ON func_api.Cod_Funcionalidade = grp_func.Cod_Funcionalidade
JOIN Tb_Autorizacao_Funcionalidade func
ON func.Cod_Funcionalidade = grp_func.Cod_Funcionalidade
JOIN Tb_Autorizacao_Api api
ON api.Id_Api = func_api.Id_Api
LEFT JOIN Tb_Autorizacao_Grupo_Usuario_Externo grp_usu
ON grp_usu.Id_Grupo = grp.Id_Grupo
LEFT JOIN Tb_Autorizacao_Usuario_Externo usu
ON grp_usu.Id_Usuario_Externo = usu.Id_Usuario_Externo
WHERE usu.Fl_Usu_Sistema = 0
AND api.Fl_Permite_Usu_Externo = 1;


CREATE VIEW View_Autorizacao_Acesso_SQL_Live   
AS    
SELECT 
	UsuSistema_ou_Cod_Grupo = UPPER(dp.name),
	'Schema' = COALESCE(se.name,so.name),
	Desc_Permissao = (perm.state_desc + ' ' + perm.permission_name),
	Desc_Objeto = obj.name,
	Desc_Tp_Objeto = 	CASE WHEN perm.class_desc = 'OBJECT_OR_COLUMN' THEN 'VIEW'
						ELSE perm.class_desc END
FROM sys.database_permissions AS perm
JOIN sys.database_principals AS dp 
ON perm.grantee_principal_id = dp.principal_id
LEFT JOIN sys.schemas AS se 
ON se.schema_id = perm.major_id
LEFT JOIN sys.objects obj 
ON obj.object_id = perm.major_id
LEFT JOIN sys.schemas AS so 
ON so.schema_id = obj.schema_id
WHERE UPPER(dp.name) not in ('DBO', 'GUEST', 'PUBLIC', 'APP_PRISMA')

CREATE VIEW [dbo].[Report_Prisma_View_Autorizacao_Acesso_SQL_Lake]   
AS   
SELECT 
	Usuario = UPPER(dp.name),
	'Schema' = COALESCE(se.name,so.name),
	Desc_Permissao = (per.state_desc + ' ' + per.permission_name),
	Desc_Objeto = COALESCE(obj.name, dpec.name),
	Desc_Tp_Objeto = 	CASE WHEN per.class_desc = 'OBJECT_OR_COLUMN' then 'VIEW'
						ELSE per.class_desc END
FROM sys.database_permissions per
LEFT JOIN sys.database_scoped_credentials dpec ON dpec.credential_id = per.major_id
LEFT JOIN sys.database_principals dp ON dp.principal_id = per.grantee_principal_id
LEFT JOIN sys.objects obj on obj.object_id = per.major_id
LEFT JOIN sys.schemas se on se.schema_id = per.major_id
LEFT JOIN sys.schemas so ON so.schema_id = obj.schema_id
WHERE UPPER(dp.name) not in ('DBO', 'GUEST', 'PUBLIC', 'APP_PRISMA')
AND per.permission_name not like 'CONTROL';
