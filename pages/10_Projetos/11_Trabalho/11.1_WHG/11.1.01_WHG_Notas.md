        {
            var tbGrupos = await _context.TbParametroConsulta
                .Where(i => i.NmReferencia == "ComplianceGrupo")
                .Select(i => new

                    i.IdUsu,
                    i.DtCpu,
                    i.DtCpuFim
                }).ToListAsync();

            // Cria uma lista para armazenar todos os grupos de `JsParametro`
            var gruposList = new List<GrupoDto>();

            // Para cada item na lista, desserialize o `JsParametro` para `List<GrupoDto>` e adicione os grupos na lista final
            foreach (var item in tbGrupos)
            {
                if (!string.IsNullOrWhiteSpace(item.JsParametro))
                {
                    var grupos = JsonConvert.DeserializeObject<List<GrupoDto>>(item.JsParametro);
                    if (grupos != null)
                    {
                        gruposList.AddRange(grupos);
                    }
                }
            }

            // Retorna a lista de grupos em formato JSON, utilizando o `DataSourceLoader`
            return Ok(gruposList);
        }

        private void PopulateRegraModel(TbCompliancePldRegra model, IDictionary values)
        {
            InterceptorUserAndDate.InterceptUserAndDate(values, this.ControllerContext?.HttpContext?.User);

            string ID_REGRA = nameof(TbCompliancePldRegra.IdRegra);
            string DESC_REGRA = nameof(TbCompliancePldRegra.DescRegra);
            string OBS_REGRA = nameof(TbCompliancePldRegra.ObsRegra);
            string ID_USU = nameof(TbCompliancePldRegra.IdUsu);
            string DT_CPU = nameof(TbCompliancePldRegra.DtCpu);

            if (values.Contains(ID_REGRA))
            {
                model.IdRegra = Convert.ToInt32(values[ID_REGRA]);
            }

            if (values.Contains(DESC_REGRA))
            {
                model.DescRegra = Convert.ToString(values[DESC_REGRA]);
            }

            if (values.Contains(OBS_REGRA))
            {
                model.ObsRegra = Convert.ToString(values[OBS_REGRA]);
            }

            if (values.Contains(ID_USU))
            {
                model.IdUsu = Convert.ToString(values[ID_USU]);
            }

            if (values.Contains(DT_CPU))
            {
                model.DtCpu = Convert.ToDateTime(values[DT_CPU]);
            }
        }


        private void PopulateModel(GrupoDto model, IDictionary values)
        {
            if (values.Contains("nome"))
            {
                model.nome = Convert.ToString(values["nome"]);
            }

            if (values.Contains("palavras"))
            {
                var palavrasStr = Convert.ToString(values["palavras"]);
                model.palavras = Regex.Split(palavrasStr, @",+")
                                      .Where(p => !string.IsNullOrWhiteSpace(p))
                                      .ToList();
            }

            if (values.Contains("id"))
            {
                model.Id = Guid.TryParse(values["id"]?.ToString(), out var guid) ? guid : (Guid?)null;
            }
        }


        private void PopulateModelParametro(ParametroDto model, IDictionary values)
        {
            if (values.Contains("nome"))
            {
                model.nome = Convert.ToString(values["nome"]);
            }

            if (values.Contains("valor"))
            {
                model.valor = Convert.ToString(values["valor"]);
            }

            if (values.Contains("id"))
            {
                model.Id = Guid.TryParse(values["id"]?.ToString(), out var guid) ? guid : (Guid?)null;
            }
        }


        private string GetFullErrorMessage(ModelStateDictionary modelState)
        {
            var messages = new List<string>();

            foreach (var entry in modelState)
            {
                foreach (var error in entry.Value.Errors)
                    messages.Add(error.ErrorMessage);
            }

            return String.Join(" ", messages);
        }
    }
