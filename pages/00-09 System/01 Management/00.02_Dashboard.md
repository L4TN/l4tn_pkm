# 🎯 L4TN PKM Dashboard

## 🔥 Active Work

### Sprints em Andamento
#+BEGIN_QUERY
{:title "Active Sprints"
 :query [:find (pull ?p [*])
         :where
         [?p :block/name ?name]
         [(clojure.string/includes? ?name "sprint")]
         [?p :block/properties ?props]
         [(get ?props :status) ?status]
         [(= ?status "active")]]
}
#+END_QUERY

### Tasks Pendentes (Todos os Projetos)
#+BEGIN_QUERY
{:title "All Open Tasks"
 :query [:find (pull ?b [*])
         :where
         [?b :block/marker ?marker]
         [(contains? #{"TODO" "DOING" "NOW"} ?marker)]]
 :collapsed? false
}
#+END_QUERY

### Tasks por Projeto
#+BEGIN_QUERY
{:title "Tasks - Impulso"
 :query [:find (pull ?b [*])
         :where
         [?b :block/marker ?marker]
         [(contains? #{"TODO" "DOING"} ?marker)]
         [?b :block/refs ?r]
         [?r :block/name "impulso"]]
}
#+END_QUERY

## 📊 Knowledge Management

### Notas Recentes (Últimos 7 Dias)
#+BEGIN_QUERY
{:title "Recent Notes"
 :query [:find (pull ?p [*])
         :where
         [?p :block/created-at ?t]
         [(> ?t ?start)]]
 :inputs [:7d-before]
}
#+END_QUERY

### Notas Órfãs (Sem Backlinks)
#+BEGIN_QUERY
{:title "Orphan Pages"
 :query [:find (pull ?p [*])
         :where
         [?p :block/name ?name]
         (not [?b :block/refs ?p])
         [(not= ?name "contents")]
         [(not= ?name "favorites")]
         [(not-empty ?name)]]
}
#+END_QUERY

## ⚠️ Maintenance Alerts

### Arquivos Problemáticos
#+BEGIN_QUERY
{:title "Files Needing Attention"
 :query [:find (pull ?p [*])
         :where
         [?p :block/name ?name]
         (or
          [(clojure.string/includes? ?name "temp")]
          [(clojure.string/includes? ?name "backup")]
          [(clojure.string/includes? ?name "draft")]
          [(clojure.string/includes? ?name "untitled")])]
}
#+END_QUERY
